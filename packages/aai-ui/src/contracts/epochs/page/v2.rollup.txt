import { ClientConfigResponse } from '@alexkroman1/aai/protocol';
import { ComponentType } from 'react';

// @public @input
type ClientTheme = {
    bg?: string;
    primary?: string;
    text?: string;
    surface?: string;
    border?: string;
};

// @public
export function fetchClientConfig(options?: FetchClientConfigOptions): Promise<ClientConfigResponse>;

// @public @input
export type FetchClientConfigOptions = {
    platformUrl?: string | undefined;
    fetch?: typeof globalThis.fetch | undefined;
    signal?: AbortSignal | undefined;
};

// @public
export function mountPage(config?: PageConfig): PageHandle;

// @public @input
export type PageConfig = {
    component?: ComponentType;
    target?: string | HTMLElement;
    name?: string;
    theme?: ClientTheme;
};

// @public @sealed
export type PageHandle = {
    dispose(): void;
    [Symbol.dispose](): void;
};
