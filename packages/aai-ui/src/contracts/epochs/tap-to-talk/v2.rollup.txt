// @public
export function useTapToTalk(options?: UseTapToTalkOptions): UseTapToTalkResult;

// @public @input
export type UseTapToTalkOptions = {
    key?: string | false | undefined;
    idleHangupMs?: number | undefined;
    thinkingHangupMs?: number | undefined;
    connectTimeoutMs?: number | undefined;
};

// @public @sealed
export type UseTapToTalkResult = {
    phase: "idle" | "connecting" | "active" | (string & {});
    live: boolean;
    failed: boolean;
    toggle: () => void;
    send: (text: string) => void;
    hangUp: () => void;
    buttonProps: {
        onClick: () => void;
        onKeyDown: (event: {
            code: string;
            preventDefault(): void;
        }) => void;
        onKeyUp: (event: {
            code: string;
            preventDefault(): void;
        }) => void;
        "aria-pressed": boolean;
    };
};
