// @public @input
export type ClientTheme = {
    bg?: string;
    primary?: string;
    text?: string;
    surface?: string;
    border?: string;
};

// @public @sealed
export type ResolvedClientTheme = {
    readonly bg: string;
    readonly primary: string;
    readonly text: string;
    readonly surface: string;
    readonly border: string;
};

// @public
export function useTheme(): ResolvedClientTheme;
