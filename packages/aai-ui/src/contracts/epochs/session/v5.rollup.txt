import type { DefaultToolResult } from '@alexkroman1/aai';
import { SessionErrorCode } from '@alexkroman1/aai/protocol';

// @public
export const AGENT_STATE_LABELS: Readonly<Record<AgentState, string>>;

// @public
export type AgentCustomEvent = {
    readonly id: number;
    readonly event: string;
    readonly data: unknown;
};

// @public
export type AgentState = "disconnected" | "connecting" | "ready" | "listening" | "thinking" | "speaking" | "error";

// @public
type AgentStateFrame = Readonly<Record<string, unknown>>;

// @public
export function browserClientId(platformUrl?: string): string;

// @public @sealed
export type BrowserSession = {
    readonly [browserSessionBrand]: true;
    getSnapshot(): SessionSnapshot;
    subscribe(callback: () => void): () => void;
    connect(options?: {
        signal?: AbortSignal;
    }): void;
    cancel(): void;
    readonly userTurn: UserTurnControls;
    readonly identity: SessionIdentity;
    sendText(text: string, options?: SendTextOptions): void;
    setMicMuted(muted: boolean): void;
    resetState(): void;
    reset(): void;
    disconnect(): void;
    start(): void;
    toggle(): void;
    end(): void;
    restart(): void;
    resume(sessionId: string): void;
    [Symbol.dispose](): void;
};

// @public
export const browserSessionBrand: unique symbol;

// @public
export type ChatMessage = {
    id: number;
    role: "user" | "assistant";
    content: string;
};

// @public
export type ConversationItem = {
    kind: "message";
    message: ChatMessage;
} | {
    kind: "tool";
    toolCall: ToolCallInfo;
};

// @public
export function createBrowserSession(options: VoiceSessionOptions): BrowserSession;

// @public
export type SendTextOptions = {
    connect?: boolean | undefined;
};

// @public
export type Session = SessionSnapshot & SessionActions;

// @public
export type SessionActions = {
    start(): void;
    cancel(): void;
    sendText(text: string, options?: SendTextOptions): void;
    setMicMuted(muted: boolean): void;
    resetState(): void;
    reset(): void;
    restart(): void;
    resume(sessionId: string): void;
    disconnect(): void;
    toggle(): void;
    end(): void;
};

// @public
export type SessionError = {
    readonly code: SessionErrorCode;
    readonly message: string;
    readonly fatal: boolean;
};

export { SessionErrorCode }

// @public @sealed
export type SessionIdentity = {
    readonly platformUrl: string;
    clientId(): string | undefined;
    holderId(): string;
    sessionId(): string | undefined;
};

// @public
export type SessionSnapshot = {
    readonly state: AgentState;
    readonly recording: boolean;
    readonly micMuted: boolean;
    readonly apiUrl: string;
    readonly contentVersion: number;
    readonly messages: ChatMessage[];
    readonly toolCalls: ToolCallInfo[];
    readonly customEvents: AgentCustomEvent[];
    readonly agentState: AgentStateFrame | null;
    readonly userTranscript: string | null;
    readonly agentTranscript: string | null;
    readonly error: SessionError | null;
    readonly started: boolean;
    readonly running: boolean;
};

// @public
type ToolCallInfo = {
    callId: string;
    name: string;
    args: Record<string, DefaultToolResult>;
    status: "pending" | "done";
    result?: string | undefined;
    seq: number;
    afterMessageId: number;
};

// @public
export function useClientId(): string | undefined;

// @public
export function useConversation(): UseConversationResult;

// @public
export type UseConversationResult = {
    items: readonly ConversationItem[];
    streaming: string | null;
    transcript: UseUserTranscriptResult;
    thinking: boolean;
};

// @public
type UserTurnControls = {
    start(): void;
    commit(): void;
    clear(): void;
};

// @public
export function useSession(): Session;

// @public
export function useSessionActions(): SessionActions;

// @public
export function useSessionControls(): UseSessionControlsResult;

// @public
export type UseSessionControlsResult = {
    started: boolean;
    running: boolean;
    start: () => void;
    toggle: () => void;
    restart: () => void;
    end: () => void;
};

// @public
export function useSessionError(): SessionError | null;

// @public
export function useSessionId(): string | undefined;

// @public
export function useSessionSelector<T>(selector: (snapshot: SessionSnapshot) => T, isEqual?: (a: T, b: T) => boolean): T;

// @public
export function useSessionStatus(): AgentState;

// @public
export function useUserTranscript(): UseUserTranscriptResult;

// @public
export interface UseUserTranscriptResult {
    partial: string | null;
    speaking: boolean;
    text: string;
}

// @public
export type VoiceSessionOptions = {
    platformUrl: string;
    onSessionId?: ((sessionId: string) => void) | undefined;
    resumeSessionId?: string | undefined;
    location?: string | (() => string | undefined) | undefined;
    phone?: string | (() => string | undefined) | undefined;
    client?: string | (() => string | undefined) | undefined;
    WebSocket?: WebSocketConstructor | undefined;
};

// @public
export type WebSocketConstructor = {
    new (url: string | URL, protocols?: string | string[]): WebSocket;
    readonly OPEN: number;
};
