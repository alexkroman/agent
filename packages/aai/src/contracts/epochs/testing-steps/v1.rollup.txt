// @public

    status: "pending" | "running";
})
/** `output` is what the workflow function returned. */
| (WorkflowRunBase & {
    status: "completed";
    output: R;
})
/** `error` is the failure message. */
| (WorkflowRunBase & {
    status: "failed";
    error: string;
})
/** Cancelled by {@link WorkflowClient.cancel}; it produced no output. */
| (WorkflowRunBase & {
    status: "cancelled";
});
