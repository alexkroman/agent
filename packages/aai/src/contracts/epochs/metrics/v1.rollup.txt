// @public
type AgentTranscriptRecovery = "turn-failed" | "session-failed" | (string & {});

// @public
export function createMetricsCollector(options?: MetricsCollectorOptions): MetricsCollector;

// @public
export type MetricsCollectedEvent = SessionEvent<"metrics.collected">;

// @public @sealed
export interface MetricsCollector {
    collect(sample: MetricsCollectedEvent | MetricsSample): void;
    reset(): void;
    summary(): MetricsSummary;
}

// @public @input
export interface MetricsCollectorOptions {
    maxSamples?: number;
}

// @public @input
export interface MetricsSample {
    interrupted: boolean;
    latencyMs?: number | undefined;
    // (undocumented)
    llm?: {
        ttftMs?: number | undefined;
        durationMs: number;
        steps: number;
        inputTokens?: number | undefined;
        outputTokens?: number | undefined;
    } | undefined;
    // (undocumented)
    stt?: {
        endpointingMs?: number | undefined;
    } | undefined;
    // (undocumented)
    tts?: {
        ttfbMs?: number | undefined;
        characters: number;
    } | undefined;
}

// @public @sealed
export interface MetricsSummary {
    interrupted: number;
    latencyMs?: MetricStat;
    llmDurationMs?: MetricStat;
    llmInputTokens: number;
    // (undocumented)
    llmOutputTokens: number;
    // (undocumented)
    llmSteps: number;
    llmTtftMs?: MetricStat;
    replies: number;
    sttEndpointingMs?: MetricStat;
    // (undocumented)
    ttsCharacters: number;
    ttsTtfbMs?: MetricStat;
}

// @public @sealed
export interface MetricStat {
    count: number;
    // (undocumented)
    max: number;
    // (undocumented)
    mean: number;
    // (undocumented)
    min: number;
    p50: number;
    p95: number;
}

// @public @sealed
interface RestoredToolCall {
    afterMessageIndex: number;
    // (undocumented)
    args: Record<string, unknown>;
    // (undocumented)
    callId: string;
    // (undocumented)
    name: string;
    // (undocumented)
    result?: string | undefined;
    status: "pending" | "done";
}

// @public
type SessionErrorCode = "stt" | "llm" | "tts" | "tool" | "protocol" | "connection" | "audio" | "internal" | (string & {});

// @public
type SessionEvent<K extends SessionEventType = SessionEventType> = SessionEventMap[K];

// @public @sealed
interface SessionEventMap {
    "agentTranscript.committed": {
        type: "agentTranscript.committed";
        meta: SessionEventMeta;
        text: string;
        recovery?: AgentTranscriptRecovery | undefined;
        recorded?: false | undefined;
    };
    "agentTranscript.updated": {
        type: "agentTranscript.updated";
        meta: SessionEventMeta;
        text: string;
    };
    "audio.completed": {
        type: "audio.completed";
        meta: SessionEventMeta;
    };
    "custom.emitted": {
        type: "custom.emitted";
        meta: SessionEventMeta;
        event: string;
        data: unknown;
    };
    // (undocumented)
    "error.reported": {
        type: "error.reported";
        meta: SessionEventMeta;
        code: SessionErrorCode;
        message: string;
        fatal: boolean;
    };
    "guardrail.blocked": {
        type: "guardrail.blocked";
        meta: SessionEventMeta;
        direction: "input" | "output" | (string & {});
        replacement: string;
    };
    "history.restored": {
        type: "history.restored";
        meta: SessionEventMeta;
        messages: {
            role: "user" | "assistant";
            content: string;
        }[];
        toolCalls: RestoredToolCall[];
    };
    "metrics.collected": {
        type: "metrics.collected";
        meta: SessionEventMeta;
        interrupted: boolean;
        latencyMs?: number | undefined;
        stt?: {
            endpointingMs?: number | undefined;
        } | undefined;
        llm?: {
            ttftMs?: number | undefined;
            durationMs: number;
            steps: number;
            inputTokens?: number | undefined;
            outputTokens?: number | undefined;
        } | undefined;
        tts?: {
            ttfbMs?: number | undefined;
            characters: number;
        } | undefined;
    };
    "provider.failedOver": {
        type: "provider.failedOver";
        meta: SessionEventMeta;
        stage: "stt" | "llm" | "tts" | (string & {});
        from: string;
        to: string;
        reason: string;
    };
    "reply.cancelled": {
        type: "reply.cancelled";
        meta: SessionEventMeta;
    };
    "reply.completed": {
        type: "reply.completed";
        meta: SessionEventMeta;
    };
    "session.configured": {
        type: "session.configured";
        meta: SessionEventMeta;
        audioFormat: string;
        sampleRate: number;
        ttsSampleRate: number;
        sessionId: string;
    };
    "session.reset": {
        type: "session.reset";
        meta: SessionEventMeta;
    };
    "session.timedOut": {
        type: "session.timedOut";
        meta: SessionEventMeta;
    };
    "speech.started": {
        type: "speech.started";
        meta: SessionEventMeta;
    };
    // (undocumented)
    "speech.stopped": {
        type: "speech.stopped";
        meta: SessionEventMeta;
    };
    "state.updated": {
        type: "state.updated";
        meta: SessionEventMeta;
        state: Record<string, unknown>;
    };
    // (undocumented)
    "tool.called": {
        type: "tool.called";
        meta: SessionEventMeta;
        toolCallId: string;
        toolName: string;
        args: Record<string, unknown>;
    };
    // (undocumented)
    "tool.completed": {
        type: "tool.completed";
        meta: SessionEventMeta;
        toolCallId: string;
        result: string;
    };
    "usage.updated": {
        type: "usage.updated";
        meta: SessionEventMeta;
        inputTokens: number;
        outputTokens: number;
        totalTokens: number;
        steps: number;
    };
    "userTranscript.committed": {
        type: "userTranscript.committed";
        meta: SessionEventMeta;
        text: string;
    };
    "userTranscript.updated": {
        type: "userTranscript.updated";
        meta: SessionEventMeta;
        text: string;
        eotConfidence?: number | undefined;
    };
    "userTurn.exceeded": {
        type: "userTurn.exceeded";
        meta: SessionEventMeta;
        limit: "words" | "duration" | (string & {});
        words: number;
        durationMs: number;
    };
}

// @public @sealed
interface SessionEventMeta {
    at: number;
    id: string;
}

// @public
type SessionEventType = Extract<keyof SessionEventMap, string>;
