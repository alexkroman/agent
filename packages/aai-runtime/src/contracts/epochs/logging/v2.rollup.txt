// @public
export type LogContext = Record<string, unknown>;

// @public
export type LogFn = (message: string, ctx?: LogContext) => void;

// @public
export interface Logger {
    // (undocumented)
    debug: LogFn;
    // (undocumented)
    error: LogFn;
    // (undocumented)
    info: LogFn;
    // (undocumented)
    warn: LogFn;
}

// @public
export type LogLevel = "info" | "warn" | "error" | "debug";
